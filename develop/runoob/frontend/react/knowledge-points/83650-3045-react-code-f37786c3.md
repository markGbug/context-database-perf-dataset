```text
class HelloMessage extends React.Component {
  render() {
    return (
      <h1>Hello, {this.props.name}</h1>
    );
  }
}

HelloMessage.defaultProps = {
  name: 'Runoob'
};

const element = <HelloMessage/>;

const root = ReactDOM.createRoot(document.getElementById("root"));
root.render(
  element
);
```
